ORG 0x6D3
ARRAY: WORD 0x0200, 0x0300, 0xFFFF, 13 DUP (?) ; 0000 0010 0000 0000
NUM_PTR: WORD $ARRAY
LENGTH: WORD 16
NUM: WORD ?
NUM32_2: WORD ? ; Двойка - это младшее слово, то-есть FFFE 1000 <-- эта часть
NUM32_1: WORD ? ; Единица - старшее, то-есть эта часть --> FFFE 1000
CHECK_SIGN_MASK: WORD 0x0200 ; Маска, чтобы оставить ранее отсчитанный последний бит
MINUS_MASK: WORD 0xFE00 ; Маска для отрицательных чисел, чтобы расширить их до 16 бит
M_TIMES: WORD 9 ; Количество итераций для умножения числа в функции
CONSTANT: WORD 328 ; Константа, которую нужно добавить
NULL_TRASH_MASK: WORD 0x3FF ; Маска для отбрасывания мусорных битов

ORG 0x700
ITER:
  CLA
  ST $NUM32_1 ; Очистим наши текущие слова
  ST $NUM32_2
  LD (NUM_PTR)+ ; Загрузим новое число
  AND $NULL_TRASH_MASK
  ST $NUM
  EXTEND_SIGN: AND $CHECK_SIGN_MASK ; Проверяем минус 10-разрядного числа
    BEQ POS ; Если минуса нет, то не расширяем знак
    OR $MINUS_MASK ; Если минус есть, то расширяем знак до всех остальных битов
    POS: OR $NUM
    ST $NUM
  CONVERT_TO_32:
    ST $NUM32_2 ; Помещаем число в младшее слово
    BPL POS32 
    LD #0xFF ; Если число отрицательное, то расширяем его до 32-бит единицами, чтобы знак сохранить
    ST $NUM32_1
    JUMP $MULTIPLY_BY_NINE 
    POS32: CLA ; Если наше число положительное, то расширяем его старшее слово нулями
      ST $NUM32_1
  MULTIPLY_BY_NINE:
    LD $NUM32_2 ; Выполним умножение на 8 при помощи чередующихся ASL и ROL
    ASL ; Умножаем наше число на 2
    ST $NUM32_2
    LD $NUM32_1
    ROL ; Сдвиг переносим на старшее слово
    ST $NUM32_1
    LD $NUM32_2
    ASL
    ST $NUM32_2
    LD $NUM32_1
    ROL
    ST $NUM32_1
    LD $NUM32_2
    ASL
    ST $NUM32_2
    LD $NUM32_1
    ROL
    ST $NUM32_1
    LD $NUM32_2 ; Вручную добавим 1 последнее число
    ADD $NUM
    ST $NUM32_2
    LD $NUM
    BPL POS_ADD
    LD $NUM32_1 ; Добавим отрицательную часть нашего M к старшему слову
    ADC #0xFF
    ST $NUM32_1
    JUMP $SUM_WITH_328
    POS_ADD: LD $NUM32_1
    ADC #0
    ST $NUM32_1
  SUM_WITH_328:
    LD $NUM32_2 ; Берём младшее слово
    ADD $CONSTANT ; Складываем с константой = 328
    ST $NUM32_2
    LD $NUM32_1
    ADC #0 ; К старшему слову прибавляем Carry Flag
    ST $NUM32_1
  JUMP $SAVE_RES ; Прыгаем на сохранение данных, иначе относительная адресация не дотягивается
  CONTINUE: LOOP $LENGTH ; Переходим к следующему элементу
  JUMP $ITER
HLT
